10:47　숫자 자료형과 결측치 확인

print(df.isna().sum())

# 창원의 2024년 6월 기온 분석 | 영수쌤의 Python 교실
# 출처: 기상청 기상자료개방포털, 창원(155), 2024-06-01~2024-06-30
# 이 파일과 changwon_2024_06.csv를 같은 폴더에 놓고 그 폴더에서 실행하세요.
import pandas as pd
import matplotlib.pyplot as plt

# 1. 원본 CSV 읽기: 앞 7줄은 설명, 8번째 줄은 열 이름입니다.
df = pd.read_csv("changwon_2024_06.csv", encoding="cp949", skiprows=7)
print("원본 크기:", df.shape)
print(df.head())

# 2. 필요한 두 열을 선택하고 간단한 이름으로 변경합니다.
df = df[["날짜", "평균기온(℃)"]].copy()
df.columns = ["날짜", "기온"]
df["날짜"] = pd.to_datetime(df["날짜"].str.strip(), errors="coerce")
df["기온"] = pd.to_numeric(df["기온"], errors="coerce")
print("원본 결측치:")
print(df.isna().sum())
df = df.dropna(subset=["날짜"]).sort_values("날짜")

# 3. 원자료와 구분한 결측치 연습용 사본을 만듭니다.
# 원자료의 실제 결측은 0개입니다. 아래 3개는 교육 목적으로 비운 값입니다.
practice = df.copy()
missing = ["2024-06-07", "2024-06-14", "2024-06-21"]
practice.loc[practice["날짜"].isin(pd.to_datetime(missing)), "기온"] = float("nan")
practice.to_csv("practice_missing.csv", index=False, encoding="utf-8-sig")

# 4. 빈칸이 있는 파일을 다시 읽고 결측치를 확인합니다.
work = pd.read_csv("practice_missing.csv", parse_dates=["날짜"])
print("연습용 결측치:", work["기온"].isna().sum())
print(work.loc[work["기온"].isna()])

# 5. 날짜를 인덱스로 두고 시간 간격을 고려하여 내부의 빈칸을 보간합니다.
work = work.set_index("날짜").sort_index()
work["보간여부"] = work["기온"].isna()
work["기온"] = work["기온"].interpolate(method="time", limit_area="inside")
work = work.reset_index()
print("처리 후 결측치:", work["기온"].isna().sum())
print(work.loc[work["보간여부"]])

# 6. 시작일과 종료일을 모두 포함하여 특정 기간을 추출합니다.
start = "2024-06-10"
end = "2024-06-20"
part = work.loc[work["날짜"].between(start, end)].copy()
original = df.loc[df["날짜"].between(start, end)].copy()
print("선택 일수:", len(part))
print("연습용 평균:", round(part["기온"].mean(), 2))
print("원자료 평균:", round(original["기온"].mean(), 2))

# 7. Windows 한글 글꼴. macOS는 AppleGothic, Linux는 설치된 한글 글꼴로 변경합니다.
plt.rcParams["font.family"] = "Malgun Gothic"
plt.rcParams["axes.unicode_minus"] = False
plt.figure(figsize=(10, 5))
plt.plot(part["날짜"], part["기온"], marker="o", label="결측치 연습용 보간 결과")
plt.plot(original["날짜"], original["기온"], linestyle="--", label="기상청 원자료")
filled = part.loc[part["보간여부"]]
plt.scatter(filled["날짜"], filled["기온"], color="red", s=90, zorder=5, label="보간한 값")
plt.title("창원 일평균기온: " + start + " ~ " + end)
plt.xlabel("날짜")
plt.ylabel("기온(℃)")
plt.xticks(rotation=45)
plt.grid(alpha=0.3)
plt.legend()
plt.tight_layout()

# 8. 표와 그림을 저장한 뒤 그림 창을 엽니다.
part.to_csv("result.csv", index=False, encoding="utf-8-sig")
plt.savefig("temperature.png", dpi=150)
print("저장 완료: result.csv, temperature.png")
plt.show()
